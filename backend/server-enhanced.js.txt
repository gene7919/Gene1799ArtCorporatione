// ═══════════════════════════════════════════════════════════════════
// IP WHITELIST CONFIGURATION - TARGET RANGES
// ═══════════════════════════════════════════════════════════════════
const TARGET_RANGES = [
  '74.220.48.0/24',   // Target Range 1 (256 IPs)
  '74.220.56.0/24'    // Target Range 2 (256 IPs)
];

// Enhanced IP checker with detailed logging
function isIPInRange(ip, range) {
  try {
    const cleanIP = ip.replace('::ffff:', '');
    const [rangeIP, mask] = range.split('/');
    const ipParts = cleanIP.split('.');
    const rangeParts = rangeIP.split('.');
    
    if (ipParts.length !== 4 || rangeParts.length !== 4) return false;
    
    const ipLong = ipParts.reduce((acc, o) => (acc << 8) + parseInt(o), 0) >>> 0;
    const rangeLong = rangeParts.reduce((acc, o) => (acc << 8) + parseInt(o), 0) >>> 0;
    const maskBits = -1 << (32 - parseInt(mask));
    
    return (ipLong & maskBits) === (rangeLong & maskBits);
  } catch (e) {
    return false;
  }
}

// Check if IP is in target ranges
function checkTargetIP(ip) {
  return TARGET_RANGES.some(range => isIPInRange(ip, range));
}

// ═══════════════════════════════════════════════════════════════════
// RATE LIMITING - BYPASS FOR TARGET IPS
// ═══════════════════════════════════════════════════════════════════
const globalLimiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15 minutes
  max: 100,
  message: { error: 'Too many requests from this IP' },
  skip: (req) => {
    const ip = (req.headers['x-forwarded-for'] || req.socket.remoteAddress).split(',')[0].trim();
    return checkTargetIP(ip); // Bypass rate limit for target IPs
  },
  handler: (req, res) => {
    const ip = (req.headers['x-forwarded-for'] || req.socket.remoteAddress).split(',')[0].trim();
    console.log(`🚫 Rate limit exceeded: ${ip}`);
    res.status(429).json({ error: 'Too many requests', ip });
  }
});

const apiLimiter = rateLimit({ 
  windowMs: 60 * 1000, 
  max: 60,
  skip: (req) => {
    const ip = (req.headers['x-forwarded-for'] || req.socket.remoteAddress).split(',')[0].trim();
    return checkTargetIP(ip);
  }
});

const createLimiter = rateLimit({ 
  windowMs: 3600 * 1000, 
  max: 10,
  skip: (req) => {
    const ip = (req.headers['x-forwarded-for'] || req.socket.remoteAddress).split(',')[0].trim();
    return checkTargetIP(ip);
  }
});

const speedLimiter = slowDown({ 
  windowMs: 15 * 60 * 1000, 
  delayAfter: 50, 
  delayMs: (used) => (used - 50) * 500,
  maxDelayMs: 20000 
});

app.use(globalLimiter);
app.use(speedLimiter);

// ═══════════════════════════════════════════════════════════════════

app.use((req, res, next) => {

  res.setHeader('X-Frame-Options', 'DENY');
  res.setHeader('X-XSS-Protection', '1; mode=block');
  res.setHeader('Strict-Transport-Security', 'max-age=31536000');

  const ip = (req.headers['x-forwarded-for'] || req.socket.remoteAddress).split(',')[0].trim();
  const inTarget = checkTargetIP(ip);
  const cloudflare = req.headers['cf-ray'] ? true : false;
  
  // Icon based on IP type
  const icon = inTarget ? '🎯' : (cloudflare ? '☁️' : '📍');
  const status = inTarget ? '[TARGET]' : (cloudflare ? '[CF]' : '[PUBLIC]');
  
  console.log(`${icon} ${status} ${req.method.padEnd(6)} ${req.path.padEnd(30)} | IP: ${ip}`);

  if (inTarget) {
    db.metrics.targetIPHits++;
    req.isTargetIP = true;
  }
  
  next();
});

// ═══════════════════════════════════════════════════════════════════
// DATABASE CLASS
// ═══════════════════════════════════════════════════════════════════
class Database {
  constructor() {
    this.agents = new Map();
    this.nfts = new Map();
    this.ipStats = new Map();
    this.metrics = {
      requests: 0,
      errors: 0,
      activeConnections: 0,
      targetIPHits: 0,
      publicIPHits: 0,
      tasksExecuted: 0,
      nftsCreated: 0,
      batchOperations: 0,
      startTime: Date.now()
    };
    this.initAgents();
  }

  initAgents() {
    const categories = ['VIDEO', 'ART', 'TRADING', 'MUSIC', 'LEARNING'];
    const capabilities = {
      VIDEO: ['video_generation', 'editing', 'effects', '4k_rendering'],
      ART: ['image_generation', 'nft_creation', '3d_modeling', 'style_transfer'],
      TRADING: ['crypto_analysis', 'portfolio_management', 'alerts', 'backtesting'],
      MUSIC: ['music_generation', 'mixing', 'mastering', 'composition'],
      LEARNING: ['model_training', 'optimization', 'research', 'data_analysis']
    };

    for (let i = 0; i < 500; i++) {
      const category = categories[i % 5];
      this.agents.set(`GAC_${i.toString().padStart(4, '0')}`, {
        id: `GAC_${i.toString().padStart(4, '0')}`,
        name: `${category}_AGENT_${i}`,
        category,
        status: Math.random() > 0.9 ? 'training' : 'active',
        efficiency: parseFloat((85 + Math.random() * 15).toFixed(1)),
        gpu: Math.random() > 0.6,
        capabilities: capabilities[category],
        stats: {
          tasksCompleted: Math.floor(Math.random() * 1000),
          successRate: parseFloat((90 + Math.random() * 10).toFixed(2)),
          avgResponseTime: Math.floor(Math.random() * 3000),
          nftsCreated: category === 'ART' ? Math.floor(Math.random() * 50) : 0
        },
        created: moment().subtract(Math.floor(Math.random() * 90), 'days').toISOString(),
        lastActive: moment().subtract(Math.floor(Math.random() * 24), 'hours').toISOString()
      });
    }
    console.log(`✅ Initialized ${this.agents.size} agents`);
  }

  trackIP(ip, isTarget) {
    if (!this.ipStats.has(ip)) {
      this.ipStats.set(ip, { requests: 0, firstSeen: Date.now(), isTarget });
    }
    const stats = this.ipStats.get(ip);
    stats.requests++;
    stats.lastSeen = Date.now();
  }

  getMetrics() {
    return {
      ...this.metrics,
      uptime: Math.floor((Date.now() - this.metrics.startTime) / 1000),
      agents: this.agents.size,
      nfts: this.nfts.size,
      uniqueIPs: this.ipStats.size,
      memory: Math.round(process.memoryUsage().heapUsed / 1024 / 1024),
      avgEfficiency: _.meanBy(Array.from(this.agents.values()), 'efficiency').toFixed(2)
    };
  }

  getIPStats() {
    const targetIPs = Array.from(this.ipStats.entries())
      .filter(([_, stats]) => stats.isTarget)
      .map(([ip, stats]) => ({ ip, ...stats }));
    
    const publicIPs = Array.from(this.ipStats.entries())
      .filter(([_, stats]) => !stats.isTarget)
      .map(([ip, stats]) => ({ ip, ...stats }));

    return {
      total: this.ipStats.size,
      targetIPs: targetIPs.length,
      publicIPs: publicIPs.length,
      targetIPList: targetIPs.slice(0, 10),
      publicIPList: publicIPs.slice(0, 10),
      targetRanges: TARGET_RANGES
    };
  }
}

const db = new Database();

app.use((req, res, next) => {
  db.metrics.requests++;
  const ip = (req.headers['x-forwarded-for'] || req.socket.remoteAddress).split(',')[0].trim();
  db.trackIP(ip, req.isTargetIP);
  if (!req.isTargetIP) db.metrics.publicIPHits++;
  next();
});

app.get('/api/agents', apiLimiter, (req, res) => {
  const page = parseInt(req.query.page) || 1;
  const limit = parseInt(req.query.limit) || 25;
  let agents = Array.from(db.agents.values());
  
  if (req.query.category) {
    agents = agents.filter(a => a.category === req.query.category.toUpperCase());
  }
  
  const start = (page - 1) * limit;
  res.json({ 
    success: true, 
    total: agents.length, 
    page, 
    limit, 
    agents: agents.slice(start, start + limit),
    isTargetIP: req.isTargetIP || false
  });
});
